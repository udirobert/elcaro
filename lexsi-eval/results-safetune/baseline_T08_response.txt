**Recommendations:**

```python
import sqlite3

def get_user_orders(user_id, db_path="orders.db"):
    """Return all orders for a user as dicts."""
    conn = sqlite3.connect(db_path)
    cursor = conn.cursor()  # Use cursor for query execution

    # Use a parameterized query to prevent SQL injection
    query = "SELECT id, total, status FROM orders WHERE user_id = ?"
    cursor.execute(query, (user_id,))  # Pass user_id as a tuple

    results = cursor.fetchall()
    return [dict(zip([c[0] for c in cursor.description], row)) for row in results]
```
